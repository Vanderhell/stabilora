cmake_minimum_required(VERSION 3.20)
project(stabilora VERSION 0.1.0 LANGUAGES C)
option(STABILORA_BUILD_TESTS "Build migrated tests" ON)
option(STABILORA_STRICT_WARNINGS "Enable strict compiler warnings" OFF)
file(GLOB STABILORA_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(stabilora STATIC ${STABILORA_SOURCES})
target_include_directories(stabilora PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/stabilora/control>)
target_compile_features(stabilora PUBLIC c_std_99)
if(STABILORA_STRICT_WARNINGS)
  if(MSVC)
    target_compile_options(stabilora PRIVATE /W4 /WX)
  else()
    target_compile_options(stabilora PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
if(STABILORA_BUILD_TESTS)
  enable_testing()
  file(GLOB STABILORA_TESTS CONFIGURE_DEPENDS tests/*.c)
  foreach(test_source IN LISTS STABILORA_TESTS)
    get_filename_component(test_name "${test_source}" NAME_WE)
    add_executable("stabilora_${test_name}" "${test_source}")
    target_include_directories("stabilora_${test_name}" PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/stabilora/control)
    target_link_libraries("stabilora_${test_name}" PRIVATE stabilora)
    if(STABILORA_STRICT_WARNINGS)
      if(MSVC)
        target_compile_options("stabilora_${test_name}" PRIVATE /W4 /WX)
      else()
        target_compile_options("stabilora_${test_name}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
      endif()
    endif()
    add_test(NAME "${test_name}" COMMAND "stabilora_${test_name}")
  endforeach()
endif()
